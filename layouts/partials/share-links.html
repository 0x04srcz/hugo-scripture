{{- /* layouts/partials/share-links.html */ -}} {{- $url := .Permalink -}} {{- $title := .Title -}} {{- /* Prefer .Description; else .Summary; else a trimmed slice of body text */ -}} {{- $desc := "" -}} {{- with .Description -}} {{- $desc = . -}} {{- else -}} {{- with .Summary -}} {{- $desc = . | plainify -}} {{- end -}} {{- end -}} {{- /* Length control: set in config; defaults to 200 chars */ -}} {{- $max := site.Params.share.summaryLength | default 200 -}} {{- if gt $max 0 -}} {{- $desc = $desc | truncate $max -}} {{- else -}} {{- $desc = "" -}} {{- end -}} {{- /* Tags to hashtags; supports tags or keywords */ -}} {{- $tags := slice -}} {{- with .Params.tags }}{{ $tags = . }}{{ end -}} {{- $hashtags := slice -}} {{- range $tags -}} {{- $t := . | plainify | htmlUnescape | lower | replaceRE `[^a-z0-9_]+` "" -}} {{- if ne $t "" -}} {{- $hashtags = $hashtags | append (printf "#%s" $t) -}} {{- end -}} {{- end -}} {{- /* Share text: title + optional description + url + hashtags */ -}} {{- $parts := slice $title -}} {{- if ne $desc "" }}{{ $parts = $parts | append $desc }}{{ end -}} {{- $parts = $parts | append $url -}} {{- if gt (len $hashtags) 0 -}} {{- $parts = $parts | append (delimit $hashtags " ") -}} {{- end -}} {{- $text := delimit $parts "\n\n" -}} {{- $encURL := $url | urlquery -}} {{- $encTitle := $title | urlquery -}} {{- $mastodonIcon := `` | safeHTML -}} {{- $bskyIcon := `` | safeHTML -}} {{- $linkedinIcon := `` | safeHTML -}} {{- $redditIcon := `` | safeHTML -}}
